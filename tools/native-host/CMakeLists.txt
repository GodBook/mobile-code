cmake_minimum_required(VERSION 3.22)
project(mobilecode_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(JNI REQUIRED COMPONENTS JVM)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SRC "${ROOT}/app/src/main/cpp")
add_executable(mc_runner "${SRC}/runner.cpp")
target_include_directories(mc_runner PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(mc_runner dl)
add_library(dt_mobile SHARED "${SRC}/jdwp_transport.cpp")
target_include_directories(dt_mobile PRIVATE ${JNI_INCLUDE_DIRS})
