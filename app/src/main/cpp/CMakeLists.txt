cmake_minimum_required(VERSION 3.22)
project(mobilecode LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_link_options(-Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)

add_library(mc_process SHARED process.cpp)
target_link_libraries(mc_process log)
add_executable(mc_runner runner.cpp)
target_link_libraries(mc_runner dl)
target_link_options(mc_runner PRIVATE -Wl,--export-dynamic-symbol=dladdr)
set_target_properties(mc_runner PROPERTIES OUTPUT_NAME libmc_runner SUFFIX .so
    RUNTIME_OUTPUT_DIRECTORY "${MC_PROGRAM_OUTPUT}/${ANDROID_ABI}")
add_library(dt_mobile SHARED jdwp_transport.cpp)

set(TS "${MC_GRAMMAR_ROOT}/tree-sitter/lib")
if(NOT EXISTS "${TS}/src/lib.c")
    message(FATAL_ERROR "Tree-sitter sources missing. Run tools/prepare_toolchains.py --grammars-only")
endif()
add_library(mc_syntax SHARED syntax.cpp "${TS}/src/lib.c")
target_include_directories(mc_syntax PRIVATE "${TS}/include" "${TS}/src")
foreach(lang c cpp python java)
    set(BASE "${MC_GRAMMAR_ROOT}/tree-sitter-${lang}/src")
    target_sources(mc_syntax PRIVATE "${BASE}/parser.c")
    if(EXISTS "${BASE}/scanner.c")
        target_sources(mc_syntax PRIVATE "${BASE}/scanner.c")
    elseif(EXISTS "${BASE}/scanner.cc")
        target_sources(mc_syntax PRIVATE "${BASE}/scanner.cc")
    endif()
    target_include_directories(mc_syntax PRIVATE "${BASE}")
endforeach()
